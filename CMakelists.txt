cmake_minimum_required(VERSION 3.20)
project(MCC_OceanHeatwave CXX)

# 强制指定 C++ 编译器为海光的 hipcc
set(CMAKE_CXX_COMPILER "hipcc")
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# 开启极度优化的编译参数 (-O3) 和 OpenMP 多线程支持 (-fopenmp)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -O3 -fopenmp")

# 寻找 OpenMP 库
find_package(OpenMP REQUIRED)

# 寻找 NetCDF-C 库 (依赖于终端 module load mathlib/netcdf/... )
find_library(NETCDF_LIB netcdf)
if(NOT NETCDF_LIB)
    message(FATAL_ERROR "未找到 NetCDF 库！请确保已执行: module load mathlib/netcdf/4.4.1/gcc-7.3.1")
endif()

# 将解耦的所有源文件联合编译为最终的可执行程序
add_executable(mcc_baseline main.cpp io_handler.cpp compute_dcu.cpp)

# 链接 OpenMP 和 NetCDF 动态库
target_link_libraries(mcc_baseline OpenMP::OpenMP_CXX ${NETCDF_LIB})
